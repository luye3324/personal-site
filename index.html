# 向量化余弦相似度矩阵计算实现 (Vectorized Cosine Similarity)
import numpy as np

def compute_cosine_similarity_matrix(rating_matrix: np.ndarray) -> np.ndarray:
    """
    输入稀疏评分矩阵 (Users x Items)，通过矩阵内积与 L2 范数广播消除原生嵌套循环:
    Sim(u, v) = (R_u · R_v) / (||R_u||_2 * ||R_v||_2 + eps)
    """
    dot_product = np.dot(rating_matrix, rating_matrix.T)
    norms = np.linalg.norm(rating_matrix, axis=1, keepdims=True)
    denominator = np.dot(norms, norms.T) + 1e-9
    similarity_matrix = dot_product / denominator
    return similarity_matrix
